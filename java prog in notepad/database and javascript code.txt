app.post('/api/orders', authenticateToken, (req, res) => {
    if (req.user.userType !== 'user') {
        return res.status(403).json({ error: 'Unauthorized' });
    }
    
    const { farmerId, items, totalAmount } = req.body;
    
    // Start transaction
    db.beginTransaction(err => {
        if (err) throw err;
        
        // Create order
        db.query(
            'INSERT INTO orders (user_id, farmer_id, total_amount) VALUES (?, ?, ?)',
            [req.user.id, farmerId, totalAmount],
            (err, result) => {
                if (err) {
                    return db.rollback(() => {
                        throw err;
                    });
                }
                
                const orderId = result.insertId;
                const orderItems = items.map(item => [orderId, item.productId, item.quantity, item.price]);
                
                // Add order items
                db.query(
                    'INSERT INTO order_items (order_id, product_id, quantity, price) VALUES ?',
                    [orderItems],
                    (err) => {
                        if (err) {
                            return db.rollback(() => {
                                throw err;
                            });
                        }
                        
                        // Update product stock
                        const updateStockQueries = items.map(item => {
                            return new Promise((resolve, reject) => {
                                db.query(
                                    'UPDATE products SET stock_quantity = stock_quantity - ? WHERE id = ?',
                                    [item.quantity, item.productId],
                                    (err) => {
                                        if (err) reject(err);
                                        else resolve();
                                    }
                                );
                            });
                        });
                        
                        Promise.all(updateStockQueries)
                            .then(() => {
                                db.commit(err => {
                                    if (err) {
                                        return db.rollback(() => {
                                            throw err;
                                        });
                                    }
                                    res.status(201).json({ message: 'Order created successfully', orderId });
                                });
                            })
                            .catch(err => {
                                db.rollback(() => {
                                    throw err;
                                });
                            });
                    }
                );
            }
        );
    });
});

// Orders API
// Create order (for customer)
app.post('/api/orders', authenticateToken, (req, res) => {
    if (req.user.userType !== 'user') {
        return res.status(403).json({ error: 'Unauthorized' });
    }
    
    const { farmerId, items, totalAmount } = req.body;
    
    // Start transaction
    db.beginTransaction(err => {
        if (err) throw err;
        
        // Create order
        db.query(
            'INSERT INTO orders (user_id, farmer_id, total_amount) VALUES (?, ?, ?)',
            [req.user.id, farmerId, totalAmount],
            (err, result) => {
                if (err) {
                    return db.rollback(() => {
                        throw err;
                    });
                }
                
                const orderId = result.insertId;
                const orderItems = items.map(item => [orderId, item.productId, item.quantity, item.price]);
                
                // Add order items
                db.query(
                    'INSERT INTO order_items (order_id, product_id, quantity, price) VALUES ?',
                    [orderItems],
                    (err) => {
                        if (err) {
                            return db.rollback(() => {
                                throw err;
                            });
                        }
                        
                        // Update product stock
                        const updateStockQueries = items.map(item => {
                            return new Promise((resolve, reject) => {
                                db.query(
                                    'UPDATE products SET stock_quantity = stock_quantity - ? WHERE id = ?',
                                    [item.quantity, item.productId],
                                    (err) => {
                                        if (err) reject(err);
                                        else resolve();
                                    }
                                );
                            });
                        });
                        
                        Promise.all(updateStockQueries)
                            .then(() => {
                                db.commit(err => {
                                    if (err) {
                                        return db.rollback(() => {
                                            throw err;
                                        });
                                    }
                                    res.status(201).json({ message: 'Order created successfully', orderId });
                                });
                            })
                            .catch(err => {
                                db.rollback(() => {
                                    throw err;
                                });
                            });
                    }
                );
            }
        );
    });
});

// Get customer orders (for customer)
app.get('/api/customer/orders', authenticateToken, (req, res) => {
    if (req.user.userType !== 'user') {
        return res.status(403).json({ error: 'Unauthorized' });
    }
    
    db.query(
        `SELECT o.*, u.name AS farmer_name 
         FROM orders o 
         JOIN users u ON o.farmer_id = u.id 
         WHERE o.user_id = ? 
         ORDER BY o.created_at DESC`,
        [req.user.id],
        (err, orders) => {
            if (err) throw err;
            
            if (orders.length === 0) {
                return res.json([]);
            }
            
            // Get order items for each order
            const orderIds = orders.map(order => order.id);
            db.query(
                `SELECT oi.*, p.name AS product_name, p.image_url 
                 FROM order_items oi 
                 JOIN products p ON oi.product_id = p.id 
                 WHERE oi.order_id IN (?)`,
                [orderIds],
                (err, orderItems) => {
                    if (err) throw err;
                    
                    // Group items by order
                    const ordersWithItems = orders.map(order => {
                        order.items = orderItems.filter(item => item.order_id === order.id);
                        return order;
                    });
                    
                    res.json(ordersWithItems);
                }
            );
        }
    );
});
