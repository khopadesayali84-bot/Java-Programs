## 1\. Database Setup (MySQL)

**File: `database_schema.sql`**

```sql
-- 1. Users Table (Farmers, Customers, and Admins)
CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    password_hash CHAR(255) NOT NULL, -- Increased size for secure hashing (e.g., password_hash)
    email VARCHAR(100) UNIQUE NOT NULL,
    full_name VARCHAR(100),
    phone_number VARCHAR(15),
    user_type ENUM('customer', 'farmer', 'admin') NOT NULL,
    upi_id VARCHAR(50), -- Only relevant for farmers/admins
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. Products Table (The items for sale)
CREATE TABLE products (
    product_id INT AUTO_INCREMENT PRIMARY KEY,
    farmer_id INT NOT NULL,
    product_name VARCHAR(100) NOT NULL,
    category ENUM('leafy veggies', 'root veggies', 'fruits', 'goods') NOT NULL,
    price DECIMAL(10, 2) NOT NULL,
    stock_quantity INT NOT NULL,
    unit VARCHAR(20) NOT NULL, -- e.g., 'kg', 'piece'
    description TEXT,
    wasted_stock INT DEFAULT 0,
    FOREIGN KEY (farmer_id) REFERENCES users(user_id)
);

-- 3. Orders and Order Details (for order history)
CREATE TABLE orders (
    order_id INT AUTO_INCREMENT PRIMARY KEY,
    customer_id INT NOT NULL,
    order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    total_amount DECIMAL(10, 2) NOT NULL,
    order_status ENUM('pending', 'delivered', 'cancelled') NOT NULL DEFAULT 'pending',
    FOREIGN KEY (customer_id) REFERENCES users(user_id)
);

CREATE TABLE order_details (
    order_detail_id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NOT NULL,
    product_id INT NOT NULL,
    quantity INT NOT NULL,
    price_at_purchase DECIMAL(10, 2) NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(order_id),
    FOREIGN KEY (product_id) REFERENCES products(product_id)
);

-- 4. Feedback Table
CREATE TABLE feedbacks (
    feedback_id INT AUTO_INCREMENT PRIMARY KEY,
    customer_id INT NOT NULL,
    feedback_text TEXT NOT NULL,
    rating INT CHECK (rating >= 1 AND rating <= 5),
    feedback_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (customer_id) REFERENCES users(user_id)
);
```

### **B. Authentication and Signup Logic**

**File: `signup.php`**

```php
<?php
require_once 'db_connect.php';

$username = $password = $email = $full_name = $user_type = $upi_id = "";
$error = "";

if ($_SERVER["REQUEST_METHOD"] == "POST") {
    $username = trim($_POST["username"]);
    $password = $_POST["password"];
    $email = trim($_POST["email"]);
    $full_name = trim($_POST["full_name"]);
    $user_type = $_POST["user_type"];
    $upi_id = ($user_type === 'farmer') ? trim($_POST["upi_id"]) : NULL;

    if (empty($username) || empty($password) || empty($email) || empty($user_type)) {
        $error = "Please fill all required fields.";
    } else {
        // Prepare a select statement to check if user already exists
        $sql = "SELECT user_id FROM users WHERE username = ? OR email = ?";
        if ($stmt = $conn->prepare($sql)) {
            $stmt->bind_param("ss", $param_username, $param_email);
            $param_username = $username;
            $param_email = $email;
            
            if ($stmt->execute()) {
                $stmt->store_result();
                if ($stmt->num_rows > 0) {
                    $error = "This username or email is already taken.";
                } else {
                    // Hash the password securely
                    $password_hash = password_hash($password, PASSWORD_DEFAULT);

                    // Prepare an insert statement
                    $sql = "INSERT INTO users (username, password_hash, email, full_name, user_type, upi_id) VALUES (?, ?, ?, ?, ?, ?)";
                    if ($stmt_insert = $conn->prepare($sql)) {
                        $stmt_insert->bind_param("ssssss", $param_user, $param_pass, $param_email, $param_name, $param_type, $param_upi);
                        
                        $param_user = $username;
                        $param_pass = $password_hash;
                        $param_email = $email;
                        $param_name = $full_name;
                        $param_type = $user_type;
                        $param_upi = $upi_id;

                        if ($stmt_insert->execute()) {
                            header("location: login.html"); // Redirect to login page
                            exit();
                        } else {
                            $error = "Something went wrong. Please try again later.";
                        }
                    }
                    $stmt_insert->close();
                }
            } else {
                $error = "Oops! Something went wrong.";
            }
        }
        $stmt->close();
    }
    $conn->close();
}
?>
```

```javascript
function toggleUpiField(userType) {
    const upiField = document.getElementById('upi_field');
    if (userType === 'farmer') {
        upiField.style.display = 'block';
        document.getElementById('upi_id').setAttribute('required', 'required');
    } else {
        upiField.style.display = 'none';
        document.getElementById('upi_id').removeAttribute('required');
    }
}
```
